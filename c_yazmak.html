# Bütün Fonksiyonları Tutabilen Cast ###########################################
Adı üstünde void* ile tanımlanan bir işaretçi doğru cast ile bütün fonksiyonlar 
için doğru bir tutucu olabilir. Burada püf nokta, bilmediğin bütün tipler için 
void* kullanmak. Zira istenen senaryoya uygun işaretçi tipine çevrilebiliyor. 
Ayrıca mimarine göre maksimum byte boyutunda olduğu için sonradan cast işlemi 
daha rahat yapılabiliyor. Kulağa ilk başta saçma geliyor ama çok fazla fonksiyon 
işaretçisi ataması yapacağınız zaman gerekliliğini anlayabiliyorsunuz. Bunun ne 
kadar işe yaradığını derleyicinin yaptığı optimizasyon ile daha rahat 
anlayabilirsiniz.
                                                       // .LC0:
                                                       //   .string "World"
                                                       // .LC1:
                                                       //   .string "Hello, %s!\n"
                                               // -O0 optimizasyon ile
    void* f = printf;                                  // movq $printf, -8(%rbp)
    ((void*(*)(void*,...))f)("Hello, %s!\n","World");  // movq -8(%rbp), %rdx
                                                       // movl $.LC0, %esi
                                                       // movl $.LC1, %edi
                                                       // movl $0, %eax
                                                       // call *%rdx
                                               // -O3 optimizasyon ile
    void* f = printf;                                  //
    ((void*(*)(void*,...))f)("Hello, %s!\n","World");  // movl $.LC0, %esi
                                                       // movl $.LC1, %edi
                                                       // xorl %eax, %eax
                                                       // jmp printf
Standartlarda aslında bu kullanıma izin verilmiyor fakat POSIX uyumluluğu için 
pratikte derleyiciler izin veriyor. C ABI'nin (System V AMD64) varlığı sayesinde 
void* ile int return type olarak aynı registere gidiyor. Burada şunu anlamanızı 
istiyorum. Fonksiyonların bu dönüş tipleri aslında bizim uydurduğumuz şeyler. 
Bunu ne kadar iyi anlarsak o kadar makineyi anlarız.

# Değiştirilebilir Fonksiyon Structları ########################################
Bir struct yaratıp sonrasında fonksiyon üzerinden özelliklerini okumak bildiğimiz 
bir pratiktir. fakat bunu değiştirebildiğimizi biliyor muydunuz?
                                                    // i.0:
                                                    //   .long 5
                                                    //   .long 0
                                                    //   .long 8
                                                    //   .long 22
    typedef struct INT {int x, y, z, w;} INT;
    INT* _INT() { 
        static INT i = {5, 0, 8, 22};
        return &i;                                  // movl $i.0, %eax
        //C23 standartı bu
        //tarz bir yazımı destekliyor
        //return &(static INT){5, 0, 8, 22};
    }

    int main() {
                                            // -O0 optimizasyon ile
        _INT()->y = 17;                             // call _INT
                                                    // movl $17, 4(%rax)
        int a = _INT()->y + 2;                      // call _INT
                                                    // movl 4(%rax), %eax
                                                    // addl $2, %eax
                                                    // movl %eax, -4(%rbp)

                                            // -O3 optimizasyon ile
        _INT()->y = 17;                             // movl $17, i.0+4(%rip)
        int a = _INT()->y + 2;

        return 0;
    }
_INT fonksiyonunu static yapmadığımız için derleyici bunu farklı dosyalarda 
çağrılabileceğini varsayıp -O3'te bile tutmayı seçiyor.

# Method Chaining ##############################################################
Nesne odaklı dilleri kullanıyorsanız bu terimi duymuş olabilirsiniz. Bunu C ile 
yapmak oldukça basittir.

    typedef struct INT {                      // her struct 4[+4]+8+8+8 byte
        int x;                                // peşinden +4 padding
        struct INT* (*set)(struct INT*, int);
        int         (*get)(struct INT*);
        struct INT* (*add)(struct INT*, int);
    } INT;
    INT* __INT_set(INT* s, int a) {s->x = a; return s;}
    int  __INT_get(INT* s)        {return s->x;}
    INT* __INT_add(INT* s, int a) {s->x += a; return s;}

    INT newINT() {return (INT){0, __INT_set, __INT_get, __INT_add};}

    int main() {
        INT i = newINT();                     // 32 byte'lık INT objesi
        
        int t = i.set(&i, 5)->add(&i, 3)->get(&i);
        
        printf("%d\n", t);
        return 0;
    }

C++ kullanırken de aslında mantık tamamen aynı çalışıyor fakat onun Class yapısı
bunu bizden gizliyor. Yani aslında C++'ta da aslında ilk argüman olarak kendini 
gönderiyor. x86 sistemlerde buna __thiscall deniyor. Farklı ABI'larda farklı 
registerlar üzerinden kendilerini gönderiyorlar.
    this pointeri:
        - __thiscall (Microsoft, x86, 32-bit):     ECX
        - Microsoft x64 çağırma kuralı (x86_64):   RCX
        - System V i386 ABI (32-bit):              stack
        - System V AMD64 ABI (x86_64):             RDI

Fakat bu yöntemi kullanarak kendimize yeni bir problem yarattık. Şimdi her INT 
yapımız 32byte yer kaplamakta. Bunların çoğu zaten aynı fonksiyonları tutmaya 
gidiyor. Her zaman daha iyi bir yolu vardır. Bunun da var:

    typedef struct __INT_VTABLE {
        struct INT *(*set)(struct INT*, int);
        int         (*get)(struct INT*);
        struct INT *(*add)(struct INT*, int);
    } __INT_VTABLE;

    typedef struct INT {
        int x;
        __INT_VTABLE *vtbl; // imzasını koyduk
    } INT;
    INT *__INT_set(INT *s, int a) {s->x = a; return s;}
    int  __INT_get(INT *s)        {return s->x;}
    INT *__INT_add(INT *s, int a) {s->x += a; return s;}

    // bunu bir kere yaratıyoruz burası çok önemli
    __INT_VTABLE __int_vtable = {__INT_set, __INT_get, __INT_add};

    INT newINT() {return (INT){0, &__int_vtable};} // adresini gönderiyoruz.

    int main() {
        INT i = newINT();

        int t = i.vtbl->set(&i, 5)->vtbl->add(&i, 3)->vtbl->get(&i);

        printf("%d\n", t);
        return 0;
    }
