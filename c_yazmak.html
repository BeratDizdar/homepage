# Compound Literal ile Dizi İşaretçisi #########################################
Pointer yardımıyla tek satırda dizi içeriğini değiştirebilirsiniz. Pointer sizin 
için sadece dizileri işaret eden bir araçtır burada.
    int* a;
    a = (int[]){142, 23};       // movl $142, -16(%rbp) 
                                // movl $23, -12(%rbp)
                                // leaq -16(%rbp), %rax
                                // movq %rax, -8(%rbp)
    a = (int[]){1, 2, 4, 5};    // movl $1, -32(%rbp)
                                // movl $2, -28(%rbp)
                                // movl $4, -24(%rbp)
                                // movl $5, -20(%rbp)
                                // leaq -32(%rbp), %rax
                                // movq %rax, -8(%rbp)

# Defect Report #76 ############################################################
Dizinin bir ötesini işaret eden pointer geçerlidir.
1993'te bu bile net değildi, komite kendi de emin değildi. C99 ile resmen çözüldü.
    int a[10];
    int* n = NULL;      // movq $0, -8(%rbp)
    int* p = &a[10];    // leaq -64(%rbp), %rax
                        // addq $40, %rax
                        // movq %rax, -8(%rbp)
    p = &*n;            // movq -16(%rbp), %rax
                        // movq %rax, -8(%rbp)

# Bütün Fonksiyonları Tutabilen Cast ###########################################
Adı üstünde void* ile tanımlanan bir işaretçi doğru cast ile bütün fonksiyonlar 
için doğru bir tutucu olabilir. Burada püf nokta, bilmediğin bütün tipler için 
void* kullanmak. Zira istenen senaryoya uygun işaretçi tipine çevrilebiliyor. 
Ayrıca mimarine göre maksimum byte boyutunda olduğu için sonradan cast işlemi 
daha rahat yapılabiliyor. Kulağa ilk başta saçma geliyor ama çok fazla fonksiyon 
işaretçisi ataması yapacağınız zaman gerekliliğini anlayabiliyorsunuz. Bunun ne 
kadar işe yaradığını derleyicinin yaptığı optimizasyon ile daha rahat 
anlayabilirsiniz.
                                                       // .LC0:
                                                       //   .string "World"
                                                       // .LC1:
                                                       //   .string "Hello, %s!\n"
    // -O0 optimizasyon ile
    void* f = printf;                                  // movq $printf, -8(%rbp)
    ((void*(*)(void*,...))f)("Hello, %s!\n","World");  // movq -8(%rbp), %rdx
                                                       // movl $.LC0, %esi
                                                       // movl $.LC1, %edi
                                                       // movl $0, %eax
                                                       // call *%rdx
    // -O3 optimizasyon ile
    void* f = printf;
    ((void*(*)(void*,...))f)("Hello, %s!\n","World");  // movl $.LC0, %esi
                                                       // movl $.LC1, %edi
                                                       // xorl %eax, %eax
                                                       // jmp printf
Standartlarda aslında bu kullanıma izin verilmiyor fakat POSIX uyumluluğu için 
pratikte derleyiciler izin veriyor. C ABI'nin (System V AMD64) varlığı sayesinde 
void* ile int return type olarak aynı registere gidiyor. Burada şunu anlamanızı 
istiyorum. Fonksiyonların bu dönüş tipleri aslında bizim uydurduğumuz şeyler. 
Bunu ne kadar iyi anlarsak o kadar makineyi anlarız.

# Değiştirilebilir Fonksiyon Structları ########################################
Bir struct yaratıp sonrasında fonksiyon üzerinden özelliklerini okumak bildiğimiz 
bir pratiktir. fakat bunu değiştirebildiğimizi biliyor muydunuz?
                                    // i.0:
                                    //   .long 5
                                    //   .long 0
                                    //   .long 8
                                    //   .long 22
    typedef struct INT {int x, y, z, w;} INT;
    INT* _INT() { 
        static INT i = {5, 0, 8, 22};
        return &i;                  // movl $i.0, %eax
        //C23 standartı bu 
        //tarz bir yazımı destekliyor
        //return &(static INT){5, 0, 8, 22};
    }

    int main() {
        // -O0 optimizasyon ile
        _INT()->y = 17;             // call _INT
                                    // movl $17, 4(%rax)
        int a = _INT()->y + 2;      // call _INT
                                    // movl 4(%rax), %eax
                                    // addl $2, %eax
                                    // movl %eax, -4(%rbp)

        // -O3 optimizasyon ile
        _INT()->y = 17;             // movl $17, i.0+4(%rip)
        int a = _INT()->y + 2;

        return 0;
    }
_INT fonksiyonunu static yapmadığımız için derleyici bunu farklı dosyalarda 
çağrılabileceğini varsayıp -O3'te bile tutmayı seçiyor.
